// Component TypeScript
export class DashboardComponent implements OnInit {
  volumeTrendData: VolumeTrendResponse[] = [];
  
  ngOnInit() {
    const startDate = new Date();
    startDate.setDate(startDate.getDate() - 14); // Last 2 weeks
    const endDate = new Date();
    
    this.analyticsService.getVolumeTrend(
      this.formatDate(startDate),
      this.formatDate(endDate)
    ).subscribe(data => {
      this.volumeTrendData = data;
      this.renderVolumeChart();
    });
  }
  
  renderVolumeChart() {
    // Chart.js / ng2-charts
    const labels = this.volumeTrendData.map(d => d.date);
    const volumeData = this.volumeTrendData.map(d => d.totalValueTraded);
    const tradeCountData = this.volumeTrendData.map(d => d.tradeCount);
    
    this.volumeChart = {
      labels: labels,
      datasets: [
        {
          label: 'Total Volume ($)',
          data: volumeData,
          borderColor: '#0066cc',
          backgroundColor: 'rgba(0, 102, 204, 0.1)',
          tension: 0.4
        },
        {
          label: 'Trade Count',
          data: tradeCountData,
          borderColor: '#ff6b6b',
          backgroundColor: 'rgba(255, 107, 107, 0.1)',
          tension: 0.4,
          yAxisID: 'y1'
        }
      ]
    };
  }
  
  private formatDate(date: Date): string {
    return date.toISOString().split('T')[0];
  }
}

export class InstrumentTrendComponent implements OnInit {
  trendData: InstrumentTrendResponse[] = [];
  chartConfig: any;
  
  ngOnInit() {
    const startDate = new Date();
    startDate.setDate(startDate.getDate() - 7); // Last week
    const endDate = new Date();
    
    this.analyticsService.getInstrumentTrend(
      this.formatDate(startDate),
      this.formatDate(endDate)
    ).subscribe(data => {
      this.trendData = data;
      this.renderMultiLineChart();
    });
  }
  
  renderMultiLineChart() {
    // Group data by ticker
    const groupedByTicker = this.groupBy(this.trendData, 'ticker');
    
    const labels = [...new Set(this.trendData.map(d => d.date))].sort();
    const colors = ['#0066cc', '#ff6b6b', '#51cf66', '#ffd93d', '#a78bfa'];
    
    const datasets = Array.from(groupedByTicker.entries()).map((entry, idx) => {
      const ticker = entry[0];
      const data = entry[1];
      
      return {
        label: ticker,
        data: labels.map(date => {
          const record = data.find(d => d.date === date);
          return record ? record.totalValue : null;
        }),
        borderColor: colors[idx % colors.length],
        backgroundColor: colors[idx % colors.length] + '20',
        tension: 0.4,
        fill: false,
        pointRadius: 4
      };
    });
    
    this.chartConfig = {
      type: 'line',
      data: {
        labels: labels,
        datasets: datasets
      },
      options: {
        responsive: true,
        interaction: { mode: 'index', intersect: false },
        plugins: {
          legend: {
            position: 'top'
          },
          title: {
            display: true,
            text: 'Instrument Trading Trends'
          }
        },
        scales: {
          y: {
            type: 'linear',
            display: true,
            title: { display: true, text: 'Trading Value ($)' }
          }
        }
      }
    };
  }
  
  private groupBy(data: any[], key: string): Map<any, any[]> {
    return data.reduce((map, item) => {
      if (!map.has(item[key])) map.set(item[key], []);
      map.get(item[key]).push(item);
      return map;
    }, new Map());
  }
  
  private formatDate(date: Date): string {
    return date.toISOString().split('T')[0];
  }
}

6. DATA AGGREGATION REFERENCE
===========================================

All trend data is aggregated daily from warehouse fact_orders table:

Query Pattern:
SELECT 
  DATE(fo.created_at) as date,
  [dimension_fields],
  COUNT(*) as trade_count,
  SUM(fo.quantity) as total_quantity,
  SUM(fo.quantity * COALESCE(fo.filled_price, fo.limit_price)) as total_value,
  AVG(COALESCE(fo.filled_price, fo.limit_price)) as average_price,
  COUNT(DISTINCT fo.account_id) as unique_traders
FROM fact_orders fo
[JOINS]
WHERE fo.status = 'FILLED' 
  AND DATE(fo.created_at) BETWEEN ? AND ?
GROUP BY DATE(fo.created_at), [dimension_fields]
ORDER BY DATE(fo.created_at) ASC

Data Warehouse Tables Used:
  - fact_orders: Core trading data (quantity, prices, timestamps)
  - dim_instruments: Security information (ticker, name, asset class)
  - dim_accounts: Account details
  - dim_clients: Client information

Only FILLED orders are included in calculations (status = 'FILLED')
All aggregates use warehouse snapshot data (not real-time)

7. PERFORMANCE CONSIDERATIONS
===========================================

Large Date Ranges:
  - Limit queries to 90 days maximum for best performance
  - Instrument trend queries may return 1000+ rows for large ranges
  - Consider pagination for UI display if needed

Sample Recommended Date Ranges:
  - Daily dashboard: Today or yesterday
  - Weekly view: Last 7 days
  - Monthly view: Last 30 days
  - Quarterly: Last 90 days

Caching Strategy:
  - Cache daily snapshots overnight
  - Refresh hourly for trending data
  - Request data once per component initialization
